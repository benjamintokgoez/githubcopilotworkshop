"""Fill-pricing path for SIMX market orders, as it ran on 2026-08-19.

This is the bounded reproduction referenced by incident QXM-4471: a resting book
served in price-time priority, a market-order sweep, and the average price that
reaches the execution-report template. It is dependency-free on purpose, so a
reproduction takes seconds rather than a full engine boot.

It is not the production engine. It is the same path, small enough to hold in
your head, and it behaves the way the confirmations in `logs/` say it behaved.
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from decimal import Decimal

ZERO = Decimal("0")

BUY = "BUY"
SELL = "SELL"


class BookError(ValueError):
    """Raised when the book is asked for something it cannot answer honestly."""


@dataclass(frozen=True)
class RestingOrder:
    """A limit order sitting on the book, waiting for someone to take it."""

    order_id: str
    side: str
    price: Decimal
    quantity: Decimal
    arrived_at: datetime


@dataclass(frozen=True)
class MarketOrder:
    """An incoming market order.

    ``indicative_price`` is the price the desk had on screen when the order was
    sent. The desk asked for it on the confirmation as context. It is not a
    limit, and the venue never guarantees it.
    """

    order_id: str
    side: str
    quantity: Decimal
    indicative_price: Decimal | None = None


@dataclass(frozen=True)
class Fill:
    """One execution against one resting order."""

    fill_id: str
    maker_order_id: str
    price: Decimal | None
    quantity: Decimal


def opposite(side: str) -> str:
    """The side a taker on ``side`` consumes."""
    if side == BUY:
        return SELL
    if side == SELL:
        return BUY
    raise BookError(f"unknown side: {side!r}")


class RestingBook:
    """Resting orders for a single symbol."""

    def __init__(self) -> None:
        self._orders: dict[str, RestingOrder] = {}
        self._remaining: dict[str, Decimal] = {}

    def add(self, order: RestingOrder) -> None:
        if order.quantity <= ZERO:
            raise BookError(f"{order.order_id}: quantity must be positive")
        if order.order_id in self._orders:
            raise BookError(f"{order.order_id}: already resting")
        if order.arrived_at.tzinfo is None:
            raise BookError(f"{order.order_id}: arrival time must be timezone-aware UTC")
        self._orders[order.order_id] = order
        self._remaining[order.order_id] = order.quantity

    def remaining(self, order_id: str) -> Decimal:
        if order_id not in self._remaining:
            raise BookError(f"{order_id}: not on the book")
        return self._remaining[order_id]

    def depth(self, side: str) -> Decimal:
        return sum(
            (self._remaining[order_id] for order_id, order in self._orders.items()
             if order.side == side),
            ZERO,
        )

    def queue(self, side: str) -> list[RestingOrder]:
        """Live orders on ``side`` in price-time priority: best price first, and
        at equal prices the earlier arrival first."""
        live = [
            order for order in self._orders.values()
            if order.side == side and self._remaining[order.order_id] > ZERO
        ]
        if side == SELL:
            return sorted(live, key=lambda order: (order.price, order.arrived_at))
        return sorted(live, key=lambda order: (-order.price, order.arrived_at))

    def reduce(self, order_id: str, quantity: Decimal) -> None:
        available = self.remaining(order_id)
        if quantity <= ZERO:
            raise BookError(f"{order_id}: cannot reduce by {quantity}")
        if quantity > available:
            raise BookError(f"{order_id}: cannot take {quantity} from {available}")
        self._remaining[order_id] = available - quantity


def match_market_order(book: RestingBook, order: MarketOrder) -> list[Fill]:
    """Sweep ``order`` against ``book`` and return the fills it produced."""
    remaining = order.quantity
    if remaining <= ZERO:
        raise BookError(f"{order.order_id}: quantity must be positive")
    fills: list[Fill] = []
    sequence = 0
    for maker in book.queue(opposite(order.side)):
        if remaining <= ZERO:
            break
        take = min(remaining, book.remaining(maker.order_id))
        sequence += 1
        fills.append(
            Fill(
                fill_id=f"F-{order.order_id}-{sequence}",
                maker_order_id=maker.order_id,
                price=order.indicative_price,
                quantity=take,
            )
        )
        book.reduce(maker.order_id, take)
        remaining -= take
    return fills


def filled_quantity(fills: list[Fill]) -> Decimal:
    """Total quantity across ``fills``."""
    return sum((fill.quantity for fill in fills), ZERO)


def average_fill_price(fills: list[Fill]) -> Decimal:
    """Quantity-weighted average price of ``fills``, as printed on the report."""
    if not fills:
        raise BookError("no fills to average")
    quantity = filled_quantity(fills)
    if quantity <= ZERO:
        raise BookError("cannot average zero quantity")
    consideration = ZERO
    for fill in fills:
        if fill.price is None:
            raise BookError(f"{fill.fill_id}: fill has no price")
        consideration += fill.price * fill.quantity
    return consideration / quantity
