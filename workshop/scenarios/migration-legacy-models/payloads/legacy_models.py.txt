"""SIMX reference-data models (legacy surface).

These models are still written against the 1.x idioms of the validation library
and are imported through its compatibility shim. See QXM-4520.

The serialised form of everything in this module is a public contract: field
names and aliases, types, the dot decimal separator in payloads, the timestamp
representation, and which inputs are rejected.
"""

from datetime import datetime, timezone
from decimal import Decimal
from typing import Optional

from pydantic.v1 import BaseModel, Field, root_validator, validator

SUPPORTED_CURRENCIES = ("EUR", "USD", "CHF", "GBP")


class InstrumentRef(BaseModel):
    """Reference data for one tradeable instrument."""

    symbol: str = Field(..., alias="sym", min_length=1, max_length=16)
    name: str = Field(..., min_length=1, max_length=80)
    currency: str = Field("EUR", min_length=3, max_length=3)
    tick_size: Decimal = Field(..., gt=0)
    venue: Optional[str] = None

    class Config:
        allow_population_by_field_name = True
        anystr_strip_whitespace = True
        json_encoders = {Decimal: str}

    @validator("symbol")
    def symbol_is_upper_case(cls, value: str) -> str:  # noqa: N805
        upper = value.upper()
        if not upper.replace(".", "").replace("-", "").isalnum():
            raise ValueError("symbol must be alphanumeric, optionally with '.' or '-'")
        return upper

    @validator("currency")
    def currency_is_supported(cls, value: str) -> str:  # noqa: N805
        upper = value.upper()
        if upper not in SUPPORTED_CURRENCIES:
            raise ValueError(f"currency must be one of {', '.join(SUPPORTED_CURRENCIES)}")
        return upper


class QuotePayload(BaseModel):
    """A two-sided quote as it is published to downstream consumers."""

    symbol: str = Field(..., alias="sym", min_length=1, max_length=16)
    bid: Decimal = Field(..., ge=0)
    ask: Decimal = Field(..., ge=0)
    as_of: datetime
    venue: Optional[str] = None

    class Config:
        allow_population_by_field_name = True
        json_encoders = {Decimal: str}

    @validator("symbol")
    def symbol_is_upper_case(cls, value: str) -> str:  # noqa: N805
        return value.upper()

    @validator("as_of")
    def as_of_is_utc(cls, value: datetime) -> datetime:  # noqa: N805
        if value.tzinfo is None:
            raise ValueError("as_of must be timezone-aware (INV-TIME-1)")
        return value.astimezone(timezone.utc)

    @root_validator(skip_on_failure=True)
    def ask_is_not_below_bid(cls, values: dict) -> dict:  # noqa: N805
        bid = values.get("bid")
        ask = values.get("ask")
        if bid is not None and ask is not None and ask < bid:
            raise ValueError("ask must not be below bid")
        return values
