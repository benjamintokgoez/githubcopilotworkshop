"""Equipment and service-rate models on a legacy compatibility shim."""

from datetime import datetime, timezone
from decimal import Decimal
from typing import Optional

from pydantic.v1 import BaseModel, Field, root_validator, validator

SUPPORTED_CURRENCIES = ("EUR", "CHF")


class EquipmentRef(BaseModel):
    asset_code: str = Field(..., alias="asset", min_length=1, max_length=24)
    name: str = Field(..., min_length=1, max_length=80)
    currency: str = Field("EUR", min_length=3, max_length=3)
    standard_hourly_rate: Decimal = Field(..., gt=0)
    service_region: Optional[str] = None

    class Config:
        allow_population_by_field_name = True
        anystr_strip_whitespace = True
        json_encoders = {Decimal: str}

    @validator("asset_code")
    def asset_code_is_upper_case(cls, value: str) -> str:  # noqa: N805
        upper = value.upper()
        if not upper.replace("-", "").isalnum():
            raise ValueError("asset code must be alphanumeric, optionally with '-'")
        return upper

    @validator("currency")
    def currency_is_supported(cls, value: str) -> str:  # noqa: N805
        upper = value.upper()
        if upper not in SUPPORTED_CURRENCIES:
            raise ValueError(f"currency must be one of {', '.join(SUPPORTED_CURRENCIES)}")
        return upper


class ServiceRatePayload(BaseModel):
    asset_code: str = Field(..., alias="asset", min_length=1, max_length=24)
    standard_rate: Decimal = Field(..., ge=0)
    emergency_rate: Decimal = Field(..., ge=0)
    as_of: datetime
    service_region: Optional[str] = None

    class Config:
        allow_population_by_field_name = True
        json_encoders = {Decimal: str}

    @validator("asset_code")
    def asset_code_is_upper_case(cls, value: str) -> str:  # noqa: N805
        return value.upper()

    @validator("as_of")
    def as_of_is_utc(cls, value: datetime) -> datetime:  # noqa: N805
        if value.tzinfo is None:
            raise ValueError("as_of must be timezone-aware (INV-TIME-1)")
        return value.astimezone(timezone.utc)

    @root_validator(skip_on_failure=True)
    def emergency_rate_is_not_lower(cls, values: dict) -> dict:  # noqa: N805
        standard = values.get("standard_rate")
        emergency = values.get("emergency_rate")
        if standard is not None and emergency is not None and emergency < standard:
            raise ValueError("emergency rate must not be below standard rate")
        return values
