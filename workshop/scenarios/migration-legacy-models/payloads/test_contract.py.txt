"""Contract checks for the MittelWerk legacy model migration."""

import json
import unittest
from datetime import UTC, datetime
from decimal import Decimal
from pathlib import Path

import legacy_api
import legacy_batch
import legacy_mcp
import legacy_models
import legacy_service
from legacy_api import equipment_document, service_rate_document, service_rate_http_json
from legacy_batch import publication_json_lines, publication_record
from legacy_models import EquipmentRef, ServiceRatePayload
from legacy_service import (
    ContractError,
    equipment_json,
    equipment_payload,
    parse_service_rate,
    service_rate_json,
    service_rate_payload,
)

HERE = Path(__file__).resolve().parent
IMPLEMENTATION_FILES = (
    "legacy_models.py",
    "legacy_service.py",
    "legacy_api.py",
    "legacy_mcp.py",
    "legacy_batch.py",
)

VALID_EQUIPMENT = {
    "asset": "press-17",
    "name": "  Hydraulic press 17  ",
    "standard_hourly_rate": "110.00",
    "service_region": "DE-SOUTH",
}

VALID_RATE = {
    "asset": "press-17",
    "standard_rate": "110.00",
    "emergency_rate": "145.00",
    "as_of": "2026-08-19T09:14:31+02:00",
}

GENERATED_AT = datetime(2026, 8, 19, 7, 15, tzinfo=UTC)


def machine_value(value: object) -> str:
    if isinstance(value, datetime):
        return value.isoformat()
    return str(value)


class ModelBoundaryTest(unittest.TestCase):
    def test_equipment_payload_keys_values_and_runtime_types(self) -> None:
        payload = equipment_payload(VALID_EQUIPMENT)
        self.assertEqual(
            list(payload),
            ["asset", "name", "currency", "standard_hourly_rate", "service_region"],
        )
        self.assertEqual(payload["asset"], "PRESS-17")
        self.assertEqual(payload["name"], "Hydraulic press 17")
        self.assertEqual(payload["currency"], "EUR")
        self.assertEqual(payload["standard_hourly_rate"], Decimal("110.00"))

    def test_optional_region_is_present_and_null_when_absent(self) -> None:
        raw = dict(VALID_EQUIPMENT)
        raw.pop("service_region")
        self.assertIsNone(equipment_payload(raw)["service_region"])

    def test_equipment_json_uses_aliases_and_dot_decimal_text(self) -> None:
        body = json.loads(equipment_json(VALID_EQUIPMENT))
        self.assertEqual(body["asset"], "PRESS-17")
        self.assertEqual(body["standard_hourly_rate"], "110.00")

    def test_rate_payload_normalises_to_aware_utc(self) -> None:
        payload = service_rate_payload(VALID_RATE)
        self.assertEqual(payload["as_of"], datetime(2026, 8, 19, 7, 14, 31, tzinfo=UTC))
        self.assertEqual(payload["standard_rate"], Decimal("110.00"))

    def test_rate_json_shape_is_unchanged(self) -> None:
        body = json.loads(service_rate_json(VALID_RATE))
        self.assertEqual(body["as_of"], "2026-08-19T07:14:31+00:00")
        self.assertEqual(body["standard_rate"], "110.00")
        self.assertEqual(body["emergency_rate"], "145.00")

    def test_round_trip_keeps_aware_utc(self) -> None:
        parsed = parse_service_rate(json.loads(service_rate_json(VALID_RATE)))
        self.assertEqual(parsed.as_of, datetime(2026, 8, 19, 7, 14, 31, tzinfo=UTC))


class AcceptedAndRejectedInputTest(unittest.TestCase):
    def test_field_name_and_alias_inputs_remain_accepted(self) -> None:
        self.assertEqual(equipment_payload(VALID_EQUIPMENT)["asset"], "PRESS-17")
        raw = dict(VALID_EQUIPMENT)
        raw["asset_code"] = raw.pop("asset")
        self.assertEqual(equipment_payload(raw)["asset"], "PRESS-17")

    def test_currency_default_and_case_normalisation_remain(self) -> None:
        raw = dict(VALID_EQUIPMENT, currency="chf")
        self.assertEqual(equipment_payload(raw)["currency"], "CHF")

    def _assert_rejected(self, raw: dict, equipment: bool) -> None:
        boundary = equipment_payload if equipment else service_rate_payload
        with self.assertRaises(ContractError) as caught:
            boundary(raw)
        self.assertTrue(str(caught.exception).strip())

    def test_zero_rate_is_rejected(self) -> None:
        self._assert_rejected(dict(VALID_EQUIPMENT, standard_hourly_rate="0"), True)

    def test_unsupported_currency_is_rejected(self) -> None:
        self._assert_rejected(dict(VALID_EQUIPMENT, currency="USD"), True)

    def test_naive_timestamp_is_rejected(self) -> None:
        self._assert_rejected(dict(VALID_RATE, as_of="2026-08-19T09:14:31"), False)

    def test_emergency_rate_below_standard_is_rejected(self) -> None:
        self._assert_rejected(dict(VALID_RATE, emergency_rate="100.00"), False)


class AdapterContractTest(unittest.TestCase):
    def test_rest_documents_keep_envelope_and_nested_model_shapes(self) -> None:
        equipment = equipment_document(
            VALID_EQUIPMENT,
            request_id="REQ-17",
            generated_at=GENERATED_AT,
        )
        rate = service_rate_document(
            VALID_RATE,
            request_id="REQ-18",
            generated_at=GENERATED_AT,
        )
        self.assertEqual(list(equipment), ["request_id", "generated_at", "equipment"])
        self.assertEqual(list(rate), ["request_id", "generated_at", "service_rate"])
        self.assertEqual(equipment["equipment"]["asset"], "PRESS-17")
        self.assertEqual(rate["service_rate"]["standard_rate"], Decimal("110.00"))
        self.assertEqual(equipment["generated_at"], "2026-08-19T07:15:00+00:00")

    def test_rest_machine_json_keeps_dot_decimals_and_utc(self) -> None:
        body = json.loads(
            service_rate_http_json(
                VALID_RATE,
                request_id="REQ-18",
                generated_at=GENERATED_AT,
            )
        )
        self.assertEqual(body["service_rate"]["standard_rate"], "110.00")
        self.assertEqual(body["service_rate"]["as_of"], "2026-08-19 07:14:31+00:00")
        self.assertNotIn(",", body["service_rate"]["standard_rate"])

    def test_mcp_text_and_structured_content_remain_equivalent(self) -> None:
        equipment = legacy_mcp.equipment_tool_result(VALID_EQUIPMENT)
        rate = legacy_mcp.service_rate_tool_result(VALID_RATE)
        self.assertEqual(
            json.loads(equipment["content"][0]["text"]),
            json.loads(json.dumps(equipment["structuredContent"], default=machine_value)),
        )
        self.assertEqual(
            json.loads(rate["content"][0]["text"]),
            json.loads(json.dumps(rate["structuredContent"], default=machine_value)),
        )
        self.assertFalse(rate["isError"])

    def test_batch_record_keeps_nested_contracts_and_asset_identity(self) -> None:
        record = publication_record(VALID_EQUIPMENT, VALID_RATE)
        self.assertEqual(
            list(record),
            ["asset", "equipment", "service_rate"],
        )
        self.assertEqual(record["asset"], "PRESS-17")
        self.assertEqual(record["equipment"]["standard_hourly_rate"], Decimal("110.00"))
        self.assertEqual(record["service_rate"]["emergency_rate"], Decimal("145.00"))

    def test_batch_json_lines_are_deterministic_machine_json(self) -> None:
        lines = publication_json_lines([(VALID_EQUIPMENT, VALID_RATE)]).splitlines()
        self.assertEqual(len(lines), 1)
        body = json.loads(lines[0])
        self.assertEqual(body["service_rate"]["standard_rate"], "110.00")
        self.assertEqual(body["service_rate"]["as_of"], "2026-08-19 07:14:31+00:00")

    def test_batch_rejects_cross_asset_publication(self) -> None:
        mismatched = dict(VALID_RATE, asset="robot-7")
        with self.assertRaisesRegex(ValueError, "assets must match"):
            publication_record(VALID_EQUIPMENT, mismatched)


class MigrationCompletedTest(unittest.TestCase):
    def test_compatibility_shim_and_deprecated_calls_are_gone(self) -> None:
        forbidden = (
            "pydantic.v1",
            ".parse_obj(",
            ".dict(",
            ".json(",
        )
        for name in IMPLEMENTATION_FILES:
            source = (HERE / name).read_text(encoding="utf-8")
            for marker in forbidden:
                self.assertNotIn(marker, source, f"{marker} remains in {name}")

    def test_models_use_supported_major_version(self) -> None:
        for model in (EquipmentRef, ServiceRatePayload):
            self.assertFalse(type(model).__module__.startswith("pydantic.v1"))

    def test_public_surface_still_exists(self) -> None:
        expected = {
            legacy_models: ("EquipmentRef", "ServiceRatePayload"),
            legacy_service: (
                "ContractError",
                "parse_equipment",
                "parse_service_rate",
                "equipment_payload",
                "equipment_json",
                "service_rate_payload",
                "service_rate_json",
            ),
            legacy_api: (
                "equipment_document",
                "service_rate_document",
                "service_rate_http_json",
            ),
            legacy_mcp: ("equipment_tool_result", "service_rate_tool_result"),
            legacy_batch: ("publication_record", "publication_json_lines"),
        }
        for module, names in expected.items():
            for name in names:
                self.assertTrue(hasattr(module, name), f"{module.__name__}.{name} missing")


if __name__ == "__main__":
    unittest.main()
