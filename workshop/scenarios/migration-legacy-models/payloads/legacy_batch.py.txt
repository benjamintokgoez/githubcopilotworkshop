"""Deterministic batch export using the published model adapters."""

from __future__ import annotations

import json
from collections.abc import Iterable
from typing import Any, Dict

from legacy_service import equipment_payload, service_rate_payload


def publication_record(
    equipment: Dict[str, Any],
    rate: Dict[str, Any],
) -> Dict[str, Any]:
    equipment_data = equipment_payload(equipment)
    rate_data = service_rate_payload(rate)
    if equipment_data["asset"] != rate_data["asset"]:
        raise ValueError("equipment and rate assets must match")
    return {
        "asset": equipment_data["asset"],
        "equipment": equipment_data,
        "service_rate": rate_data,
    }


def publication_json_lines(
    publications: Iterable[tuple[Dict[str, Any], Dict[str, Any]]],
) -> str:
    records = [publication_record(equipment, rate) for equipment, rate in publications]
    return "\n".join(
        json.dumps(record, separators=(",", ":"), default=str)
        for record in records
    )
