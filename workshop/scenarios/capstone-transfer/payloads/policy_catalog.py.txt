"""Read-only policy catalogue over the metaclass registry."""

from __future__ import annotations

from policy_base import BasePolicy


def policy_catalog() -> list[dict[str, str]]:
    return [
        {
            "name": name,
            "implementation": f"{policy_type.__module__}.{policy_type.__name__}",
        }
        for name, policy_type in sorted(BasePolicy.registered().items())
    ]
