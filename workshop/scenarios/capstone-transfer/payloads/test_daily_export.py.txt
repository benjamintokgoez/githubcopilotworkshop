"""Acceptance checks for the daily export utility (OPS-2026-118).

The expected values are the ones printed in the lab and in `acceptance.md`. The
two daylight-saving dates are not edge cases here, they are the specification.

Run from the repository root:

    python scripts/workshop.py verify capstone-transfer
"""

import json
import unittest
from datetime import UTC, date, datetime
from decimal import Decimal
from pathlib import Path

from daily_export import (
    display_total,
    export_filename,
    records_in_window,
    selection_window,
    total_of,
)

SAMPLE = Path(__file__).resolve().parent.parent / "data" / "records_2026-08-19.json"


def utc(text: str) -> datetime:
    return datetime.fromisoformat(text.replace("Z", "+00:00")).astimezone(UTC)


def sample_records() -> list[dict[str, str]]:
    with SAMPLE.open(encoding="utf-8") as handle:
        return list(json.load(handle)["records"])


class SelectionWindowTest(unittest.TestCase):
    """A business day is a local day, converted to UTC."""

    def test_spring_forward_day_is_23_hours(self) -> None:
        start, end = selection_window(date(2026, 3, 29))
        self.assertEqual(start, utc("2026-03-28T23:00:00Z"))
        self.assertEqual(end, utc("2026-03-29T22:00:00Z"))
        self.assertEqual((end - start).total_seconds(), 23 * 3600)

    def test_ordinary_summer_day_is_24_hours(self) -> None:
        start, end = selection_window(date(2026, 8, 19))
        self.assertEqual(start, utc("2026-08-18T22:00:00Z"))
        self.assertEqual(end, utc("2026-08-19T22:00:00Z"))
        self.assertEqual((end - start).total_seconds(), 24 * 3600)

    def test_fall_back_day_is_25_hours(self) -> None:
        start, end = selection_window(date(2026, 10, 25))
        self.assertEqual(start, utc("2026-10-24T22:00:00Z"))
        self.assertEqual(end, utc("2026-10-25T23:00:00Z"))
        self.assertEqual((end - start).total_seconds(), 25 * 3600)

    def test_window_bounds_are_timezone_aware_utc(self) -> None:
        for bound in selection_window(date(2026, 8, 19)):
            self.assertIsNotNone(bound.tzinfo)
            self.assertEqual(bound.utcoffset().total_seconds(), 0)


class WindowMembershipTest(unittest.TestCase):
    """The interval is half-open: the start belongs, the end does not."""

    def test_selected_records_for_the_sample_business_date(self) -> None:
        selected = records_in_window(sample_records(), date(2026, 8, 19))
        self.assertEqual(
            [record["record_id"] for record in selected],
            ["R-0002", "R-0003", "R-0004", "R-0005", "R-0006"],
        )

    def test_record_exactly_at_the_window_end_belongs_to_the_next_day(self) -> None:
        selected = records_in_window(sample_records(), date(2026, 8, 19))
        self.assertNotIn("R-0007", [record["record_id"] for record in selected])
        next_day = records_in_window(sample_records(), date(2026, 8, 20))
        self.assertIn("R-0007", [record["record_id"] for record in next_day])

    def test_record_before_the_window_start_is_excluded(self) -> None:
        selected = records_in_window(sample_records(), date(2026, 8, 19))
        self.assertNotIn("R-0001", [record["record_id"] for record in selected])

    def test_no_record_is_lost_or_counted_twice_across_three_days(self) -> None:
        all_ids = [record["record_id"] for record in sample_records()]
        seen: list[str] = []
        for day in (date(2026, 8, 18), date(2026, 8, 19), date(2026, 8, 20)):
            seen.extend(record["record_id"] for record in records_in_window(sample_records(), day))
        self.assertEqual(sorted(seen), sorted(all_ids))


class FilenameTest(unittest.TestCase):
    """Identifiers and filenames use ISO 8601 dates."""

    def test_filename_for_the_sample_business_date(self) -> None:
        self.assertEqual(export_filename(date(2026, 8, 19)), "daily_export_2026-08-19.csv")

    def test_filenames_sort_chronologically(self) -> None:
        names = [
            export_filename(date(2026, 10, 25)),
            export_filename(date(2026, 3, 29)),
            export_filename(date(2026, 8, 19)),
        ]
        self.assertEqual(sorted(names), [names[1], names[2], names[0]])


class DisplayTotalTest(unittest.TestCase):
    """Display uses a decimal comma and carries its currency."""

    def test_display_total_formats_for_a_german_reader(self) -> None:
        self.assertEqual(display_total(Decimal("1234567.891")), "1.234.567,89 EUR")

    def test_currency_is_explicit_and_configurable(self) -> None:
        self.assertEqual(display_total(Decimal("0.5"), "CHF"), "0,50 CHF")

    def test_sample_day_total_is_formatted_from_the_stored_value(self) -> None:
        selected = records_in_window(sample_records(), date(2026, 8, 19))
        total = total_of(selected)
        self.assertEqual(total, Decimal("1234567.891"))
        self.assertEqual(display_total(total), "1.234.567,89 EUR")


if __name__ == "__main__":
    unittest.main()
