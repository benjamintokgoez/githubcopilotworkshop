"""Capacity matching for the service-assignment incident."""

from __future__ import annotations

from datetime import UTC, datetime
from decimal import Decimal

from capacity_queue import CapacityQueue
from dispatch_models import (
    DispatchError,
    ServiceAssignment,
    ServiceRequest,
    ZERO,
)


class DispatchEngine:
    """Match requests against provider offers at rate-time priority."""

    def __init__(self, queue: CapacityQueue) -> None:
        self._queue = queue

    def dispatch(
        self,
        request: ServiceRequest,
        *,
        assigned_at: datetime | None = None,
    ) -> list[ServiceAssignment]:
        remaining = request.requested_hours
        assignments: list[ServiceAssignment] = []
        moment = (assigned_at or datetime.now(UTC)).astimezone(UTC)

        for offer in self._queue.eligible(request.maximum_hourly_rate):
            if remaining <= ZERO:
                break
            assigned = min(remaining, self._queue.remaining(offer.offer_id))
            assignments.append(
                ServiceAssignment(
                    assignment_id=f"SA-{request.work_order_id}-{len(assignments) + 1}",
                    work_order_id=request.work_order_id,
                    provider_code=offer.provider_code,
                    provider_offer_id=offer.offer_id,
                    hourly_rate=offer.hourly_rate,
                    assigned_hours=assigned,
                    assigned_at=moment,
                )
            )
            self._queue.reduce(offer.offer_id, assigned)
            remaining -= assigned
        return assignments


def assigned_hours(assignments: list[ServiceAssignment]) -> Decimal:
    return sum((assignment.assigned_hours for assignment in assignments), ZERO)


def average_service_rate(assignments: list[ServiceAssignment]) -> Decimal:
    if not assignments:
        raise DispatchError("no assignments to average")
    hours = assigned_hours(assignments)
    if hours <= ZERO:
        raise DispatchError("cannot average zero assigned hours")
    cost = sum(
        (
            assignment.hourly_rate * assignment.assigned_hours
            for assignment in assignments
        ),
        ZERO,
    )
    return cost / hours
